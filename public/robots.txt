# CaterKart — https://caterkart.in
#
# Two policies in one file. Search crawlers are welcome on the public pages and
# kept off everything transactional. AI crawlers are split by what they do:
# the ones that answer questions with a citation are allowed, and the ones that
# only harvest training data are not.

# ── Search engines ──────────────────────────────────────────────────────────
User-agent: *
Allow: /$
Allow: /caterbox
Allow: /our-kitchen
Allow: /about
Allow: /contact
Allow: /privacy-policy
Allow: /terms
Allow: /refund-policy
Allow: /shipping-policy
Allow: /cookie-policy
Allow: /grievance-redressal

# Bearer-token URLs. An indexed invite link IS a leaked credential: the token
# in the path is the only thing authorising a guest to read the party's guest
# list and browse menus on the host's Swiggy account. This is a security
# control, not a crawl preference.
Disallow: /invite/
Disallow: /work/join/

# Private and transactional. Nothing here is useful to a searcher, and some of
# it is somebody's order.
Disallow: /login
Disallow: /create
Disallow: /group
Disallow: /addons
Disallow: /confirm
Disallow: /event
Disallow: /settle
Disallow: /orders
Disallow: /checkout
Disallow: /create-menu
Disallow: /track-order
Disallow: /admin
Disallow: /vendor
Disallow: /work
Disallow: /api/
